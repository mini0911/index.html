// ========================================== // 1. 遊戲資料庫 (直接沿用之前的 JSON 結構) // ========================================== const BobaDatabase = { ingredients: [ { id: "tea_assam", name: "阿薩姆紅茶", category: "BaseTea", cost: 10, teaFlavor: 25, fullness: 5, appearance: 10, coolness: 15 }, { id: "tea_four_seasons", name: "四季春青茶", category: "BaseTea", cost: 12, teaFlavor: 30, fullness: 5, appearance: 15, coolness: 25 }, { id: "top_boba", name: "黑糖珍珠", category: "Topping", cost: 15, teaFlavor: 5, fullness: 35, appearance: 25, coolness: 0 }, { id: "flav_milk", name: "小農鮮奶", category: "Flavor", cost: 20, teaFlavor: 15, fullness: 25, appearance: 20, coolness: 10 } ], combos: [ { teaId: "tea_assam", toppingId: "top_boba", flavorId: "flav_milk", resultDrinkName: "黑糖珍珠鮮奶茶", synergyRating: "SuperMatch", priceMultiplier: 1.5, bonusStats: { teaFlavor: 20, fullness: 25, appearance: 30, coolness: 0 } } ] }; // ========================================== // 2. 遊戲核心狀態與存檔系統 (localStorage) // ========================================== class BobaGameEngine { constructor() { this.state = this.loadGame() || { money: 500, // 新台幣 inspiration: 30, // 💡靈感點數 menu: [], // 已研發上架的飲品 customerLevels: { // 客群等級 Student: { level: 1, exp: 0 } } }; } // 核心方法：飲品研發配對 (對應 C# 的 ResearchNewDrink) researchDrink(teaId, toppingId = "", flavorId = "") { if (this.state.inspiration < 10) { return { success: false, msg: "靈感點數不足（需要 10 點）！" }; } const tea = BobaDatabase.ingredients.find(i => i.id === teaId); const topping = BobaDatabase.ingredients.find(i => i.id === toppingId); const flavor = BobaDatabase.ingredients.find(i => i.id === flavorId); if (!tea) return { success: false, msg: "請選擇茶底！" }; this.state.inspiration -= 10; // 累加基礎數值 let baseCost = tea.cost + (topping?.cost || 0) + (flavor?.cost || 0); let stats = { teaFlavor: tea.teaFlavor + (topping?.teaFlavor || 0) + (flavor?.teaFlavor || 0), fullness: tea.fullness + (topping?.fullness || 0) + (flavor?.fullness || 0), appearance: tea.appearance + (topping?.appearance || 0) + (flavor?.appearance || 0), coolness: tea.coolness + (topping?.coolness || 0) + (flavor?.coolness || 0) }; // 檢查相性組合 const matched = BobaDatabase.combos.find( c => c.teaId === teaId && (c.toppingId || "") === toppingId && (c.flavorId || "") === flavorId ); let newDrink = {}; if (matched) { newDrink = { name: matched.resultDrinkName, rating: matched.synergyRating, stats: { teaFlavor: stats.teaFlavor + matched.bonusStats.teaFlavor, fullness: stats.fullness + matched.bonusStats.fullness, appearance: stats.appearance + matched.bonusStats.appearance, coolness: stats.coolness + matched.bonusStats.coolness } }; const total = Object.values(newDrink.stats).reduce((a, b) => a + b, 0); newDrink.sellPrice = Math.round((baseCost * 1.5 + total * 0.25) * matched.priceMultiplier); } else { const prefix = (flavor ? flavor.name : "") + (topping ? topping.name : ""); newDrink = { name: `${prefix}${tea.name}`, rating: "Normal", stats: stats }; const total = Object.values(stats).reduce((a, b) => a + b, 0); newDrink.sellPrice = Math.round(baseCost * 1.5 + total * 0.2); } this.state.menu.push(newDrink); this.saveGame(); return { success: true, drink: newDrink }; } // 自動存檔至瀏覽器 saveGame() { localStorage.setItem("boba_shop_save_v1", JSON.stringify(this.state)); } loadGame() { const raw = localStorage.getItem("boba_shop_save_v1"); return raw ? JSON.parse(raw) : null; } }
