""" 《末日便利商店》(Doomsday Convenience Store) - 核心數值與經濟/拾荒模擬腳本 包含： 1. 動態物價波動模型（基礎價值、供需、事件驅動、噪音、玩家標價購買意向） 2. 拾荒委託風險與回報模擬（角色屬性、裝備加成、區域難度、三元結算） 3. 30 天營運與市場模擬主迴圈 """ import random from dataclasses import dataclass from typing import List, Dict, Optional # ========================================== # 1. 物品與物價模型 # ========================================== @dataclass class Item: name: str category: str # Vital, Medical, Defense, Luxury base_price: float # P0 基準價 elasticity: float # 價格敏感度 alpha (如 Vital=2.0, Luxury=0.8) ITEMS = { "water": Item("純水", "Vital", 10.0, 2.0), "food": Item("乾糧", "Vital", 15.0, 1.8), "rad_pill": Item("碘片", "Medical", 40.0, 1.5), "ammo_9mm": Item("9mm子彈", "Defense", 15.0, 1.2), "cigarettes": Item("香菸條", "Luxury", 120.0, 0.8), } @dataclass class MarketEvent: name: str affected_categories: Dict[str, float] # 類別 -> 加成倍率 duration: int EVENT_POOL = [ MarketEvent("酸雨風暴", {"Medical": 2.8, "Vital": 1.2}, duration=3), MarketEvent("淨水器故障", {"Vital": 3.2}, duration=2), MarketEvent("暴民劫掠警報", {"Defense": 2.2, "Vital": 1.1}, duration=3), MarketEvent("拾荒大篷車抵達", {"Luxury": 0.6, "Defense": 0.8}, duration=1), MarketEvent("避難所狂歡夜", {"Luxury": 2.5}, duration=2), ] class PriceEngine: def __init__(self): self.active_event: Optional[MarketEvent] = None self.event_days_left: int = 0 self.supply_multipliers: Dict[str, float] = {k: 1.0 for k in ITEMS} def update_daily_event(self, day: int) -> Optional[str]: if self.event_days_left > 0: self.event_days_left -= 1 if self.event_days_left == 0: self.active_event = None return "事件結束，市場逐漸恢復常態。" return f"事件持續中：【{self.active_event.name}】（剩餘 {self.event_days_left} 天）" # 35% 機率觸發新事件 if random.random() < 0.35: self.active_event = random.choice(EVENT_POOL) self.event_days_left = self.active_event.duration return f"★ 突發事件觸發：【{self.active_event.name}】！預計持續 {self.event_days_left} 天。" return None def calculate_market_price(self, item_key: str) -> float: """計算每日市場公道價：P_t = P_0 * M_env * M_event * (1 + epsilon)""" item = ITEMS[item_key] m_env = self.supply_multipliers.get(item_key, 1.0) m_event = 1.0 if self.active_event and item.category in self.active_event.affected_categories: m_event = self.active_event.affected_categories[item.category] epsilon = random.uniform(-0.08, 0.08) price = item.base_price * m_env * m_event * (1 + epsilon) return round(max(price, 1.0), 1) def calculate_buy_chance(self, item_key: str, set_price: float, market_price: float) -> float: """根據玩家定價與市場公道價計算顧客購買意向率""" if set_price <= market_price: return 1.0 item = ITEMS[item_key] diff_ratio = (set_price - market_price) / market_price penalty = item.elasticity * (diff_ratio ** 2) return max(0.0, round(1.0 - penalty, 2)) # ========================================== # 2. 拾荒委託系統 # ========================================== @dataclass class Scavenger: name: str phy: int # 體魄 1-10 per: int # 感知 1-10 grt: int # 膽量 1-10 daily_wage: float = 20.0 profit_cut: float = 0.20 # 抽成 20% is_alive: bool = True is_injured: bool = False @dataclass class Zone: name: str days: int danger_coeff: float # 基礎危險係數 D (0.0 ~ 1.0) loot_range: tuple # (min_caps, max_caps) loot_items: List[str] ZONES = { "supermarket": Zone("廢棄超商區", 1, 0.15, (80, 150), ["water", "food", "rad_pill"]), "scrapyard": Zone("市郊廢車場", 2, 0.35, (250, 450), ["ammo_9mm", "food", "cigarettes"]), "checkpoint": Zone("軍事檢查站", 2, 0.55, (600, 1000), ["ammo_9mm", "rad_pill", "cigarettes"]), "biolab": Zone("生化實驗室遺跡", 3, 0.75, (1500, 2800), ["rad_pill", "ammo_9mm", "water", "cigarettes"]) } class ExpeditionManager: @staticmethod def calculate_death_risk(scavenger: Scavenger, zone: Zone, armor_val: int = 5, has_supplies: bool = True) -> float: """ R_death = max(5%, D - (護甲*2% + PHY*1.5% + GRT*1%)) 缺乏補給時危險增加 20% """ gear_bonus = (armor_val * 0.02) + (scavenger.phy * 0.015) + (scavenger.grt * 0.01) supply_penalty = 0.0 if has_supplies else 0.20 risk = zone.danger_coeff + supply_penalty - gear_bonus return round(max(0.05, min(0.95, risk)), 3) @classmethod def execute_expedition(cls, scavenger: Scavenger, zone: Zone, armor_val: int = 5, has_supplies: bool = True) -> dict: death_risk = cls.calculate_death_risk(scavenger, zone, armor_val, has_supplies) roll = random.random() # 結果三元分支 if roll <= death_risk: scavenger.is_alive = False return { "status": "殉職失聯", "risk": death_risk, "roll": round(roll, 3), "loot_value": 0, "loot_items": [], "wage_paid": 0, "net_profit": 0, "msg": f"☠️ {scavenger.name} 在【{zone.name}】遭遇致命伏擊，不幸殉職，裝備與物資全部遺失！" } elif roll <= death_risk + 0.25: scavenger.is_injured = True base_loot = random.randint(*zone.loot_range) * random.uniform(0.5, 0.7) chosen_item = random.choice(zone.loot_items) medical_cost = 45.0 wage = scavenger.daily_wage * zone.days net = base_loot * (1 - scavenger.profit_cut) - wage - medical_cost return { "status": "負傷返還", "risk": death_risk, "roll": round(roll, 3), "loot_value": round(base_loot, 1), "loot_items": [chosen_item], "wage_paid": wage, "medical_cost": medical_cost, "net_profit": round(net, 1), "msg": f"⚠️ {scavenger.name} 身負重傷逃回，丟失部分物資，帶回價值 {round(base_loot, 1)} 瓶蓋，需要治療！" } else: base_loot = random.randint(*zone.loot_range) extra_drops = random.sample(zone.loot_items, k=min(len(zone.loot_items), 1 + (scavenger.per // 4))) wage = scavenger.daily_wage * zone.days scav_cut = base_loot * scavenger.profit_cut net = base_loot - scav_cut - wage return { "status": "凱旋大捷", "risk": death_risk, "roll": round(roll, 3), "loot_value": base_loot, "loot_items": extra_drops, "wage_paid": wage, "scav_cut": round(scav_cut, 1), "net_profit": round(net, 1), "msg": f"🎉 {scavenger.name} 完好無損歸來！滿載而歸，總估值 {base_loot} 瓶蓋！" } # ========================================== # 3. 30 天快速模擬運算展示 # ========================================== def run_simulation(): engine = PriceEngine() player_caps = 300.0 print("=" * 65) print(" 🏪 《末日便利商店》30 天經濟與拾荒模擬引擎啟動") print(f" 初始流動資金: {player_caps} 瓶蓋 (Caps)") print("=" * 65) team = [ Scavenger("老雷 (老兵)", phy=7, per=5, grt=8, daily_wage=25.0), Scavenger("阿雅 (偵察兵)", phy=4, per=9, grt=6, daily_wage=22.0) ] for day in range(1, 31): print(f"\n--- [第 {day:02d} 天] ---") event_info = engine.update_daily_event(day) if event_info: print(f"📢 {event_info}") prices = {k: engine.calculate_market_price(k) for k in ITEMS} price_str = " | ".join([f"{ITEMS[k].name}: {prices[k]}C" for k in prices]) print(f"💰 今日牌價 -> {price_str}") if day % 3 == 1: active_scavs = [s for s in team if s.is_alive] if active_scavs: scav = random.choice(active_scavs) target_zone = ZONES["supermarket"] if player_caps < 600 else ZONES["scrapyard"] if player_caps > 1200: target_zone = ZONES["checkpoint"] print(f"🎒 派遣隊員 [{scav.name}] 前往【{target_zone.name}】探索 ({target_zone.days}天)...") res = ExpeditionManager.execute_expedition(scav, target_zone, armor_val=6, has_supplies=True) print(f" {res['msg']}") print(f" [檢定判定: 擲骰 {res['roll']} vs 死亡閾值 {res['risk']}]") player_caps += res["net_profit"] print(f" 委託淨結算收益: {res['net_profit']:+} 瓶蓋 | 當前金庫: {round(player_caps, 1)} 瓶蓋") else: print("❌ 所有拾荒隊員均已殉職，無法派遣！") print("\n" + "=" * 65) print(f"🏁 30 天模擬結束！最終金庫結餘: {round(player_caps, 1)} 瓶蓋") print("=" * 65) if __name__ == "__main__": run_simulation()
